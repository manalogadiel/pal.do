cmake_minimum_required(VERSION 3.22.1)
project(paldo_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.cache/android-ai/llama.cpp-f2cc7282ce0042641df55e0e563b75e824e2b647")
if(NOT EXISTS "${ENGINE_ROOT}/.ready")
  message(FATAL_ERROR "Run scripts/prepare-android-ai.ps1 before building Android AI")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_SUBPROCESS GGML_OPENMP GGML_NATIVE GGML_LLAMAFILE GGML_BACKEND_DL GGML_CPU_KLEIDIAI)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
# Dot-product + fp16 math: about 2x faster replies on a Galaxy A32 than generic armv8 code (17s → 9s).
# NativeEngine.cpuSupported() keeps phones without these instructions from loading it.
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
add_subdirectory("${ENGINE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(paldo-ai SHARED paldo-ai.cpp)
target_include_directories(paldo-ai PRIVATE "${ENGINE_ROOT}/include" "${ENGINE_ROOT}/vendor")
target_link_libraries(paldo-ai PRIVATE llama android log)
target_link_options(paldo-ai PRIVATE "-Wl,-z,max-page-size=16384")
